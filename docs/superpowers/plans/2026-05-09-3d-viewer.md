> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `sky_collection_card.html` に Three.js GLTFビューアを統合し、右下セレクターで「現在（画像）/ 案A（3D置換）/ 案B（モーダル3D）」を切り替えられるようにする。

**Tech Stack:** Three.js 0.165.0 (CDN), GLTFLoader, OrbitControls, vanilla JS/CSS

```js
// ─── Three.js Lazy Loader ────────────────────────────────────
let _threeCache = null;
async function loadThree() {
  if (_threeCache) return _threeCache;
  const [THREE, { GLTFLoader }, { OrbitControls }] = await Promise.all([
    import('three'),
    import('three/addons/loaders/GLTFLoader.js'),
    import('three/addons/controls/OrbitControls.js'),
  ]);
  _threeCache = { THREE, GLTFLoader, OrbitControls };
  return _threeCache;
}
```

- [ ] **Step 3: `planA_init()` 非同期関数を追加する**

```js
async function planA_init() {
  const { THREE, GLTFLoader, OrbitControls } = await loadThree();
  const canvas = document.getElementById('canvas-plan-a');
  const { w, h } = getCardDims();

  planAScene = new THREE.Scene();
  planACamera = new THREE.PerspectiveCamera(45, w / h, 0.01, 1000);

  // Lights
  planAScene.add(new THREE.AmbientLight(0xffffff, 1.2));
  const dir = new THREE.DirectionalLight(0xffffff, 1.8);
  dir.position.set(3, 5, 4);
  planAScene.add(dir);

  // Renderer
  planARenderer = new THREE.WebGLRenderer({ canvas, antialias: true });
  planARenderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
  planARenderer.setSize(w, h);
  planARenderer.setClearColor(0x0a1020, 1);

  // OrbitControls
  planAControls = new OrbitControls(planACamera, canvas);
  planAControls.enableDamping = true;
  planAControls.dampingFactor = 0.06;
  planAControls.enablePan = false;

  // Load GLTF
  const loader = new GLTFLoader();
  loader.load(players[0].modelSrc, gltf => {
    const box = new THREE.Box3().setFromObject(gltf.scene);
    const center = box.getCenter(new THREE.Vector3());
    const size = box.getSize(new THREE.Vector3());
    const maxDim = Math.max(size.x, size.y, size.z);
    gltf.scene.position.sub(center);
    planACamera.position.set(0, 0, maxDim * 2.2);
    planAControls.update();
    planAScene.add(gltf.scene);
  });

  // Animation loop
  function animate() {
    planAAnimId = requestAnimationFrame(animate);
    planAControls.update();
    planARenderer.render(planAScene, planACamera);
  }
  animate();

  // Resize handler
  window.addEventListener('resize', () => {
    if (viewMode !== 'plan-a' || !planARenderer) return;
    const { w: nw, h: nh } = getCardDims();
    planARenderer.setSize(nw, nh);
    planACamera.aspect = nw / nh;
    planACamera.updateProjectionMatrix();
  });
}
```

```js
let modalRenderer = null, modalControls = null, modalAnimId = null;

async function openModal3d(idx) {
  const p = (idx >= 0 && idx < players.length) ? players[idx] : { name: '&You', modelSrc: '3d/3d_sample_hal.gltf' };
  document.getElementById('modal3dName').textContent = p.name;

  const modal = document.getElementById('modal-3d');
  modal.classList.add('open');

  const { THREE, GLTFLoader, OrbitControls } = await loadThree();
  const canvas = document.getElementById('modal-3d-canvas');

  // キャンバスの表示サイズを取得（CSS で決まる）
  const rect = canvas.getBoundingClientRect();
  const w = rect.width || 400;
  const h = rect.height || 400;

  const scene = new THREE.Scene();
  const camera = new THREE.PerspectiveCamera(45, w / h, 0.01, 1000);

  scene.add(new THREE.AmbientLight(0xffffff, 1.2));
  const dir = new THREE.DirectionalLight(0xffffff, 1.8);
  dir.position.set(3, 5, 4);
  scene.add(dir);

  modalRenderer = new THREE.WebGLRenderer({ canvas, antialias: true });
  modalRenderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
  modalRenderer.setSize(w, h);
  modalRenderer.setClearColor(0x04040f, 1);

  modalControls = new OrbitControls(camera, canvas);
  modalControls.enableDamping = true;
  modalControls.dampingFactor = 0.06;
  modalControls.enablePan = false;

  const loader = new GLTFLoader();
  loader.load(p.modelSrc, gltf => {
    const box = new THREE.Box3().setFromObject(gltf.scene);
    const center = box.getCenter(new THREE.Vector3());
    const size = box.getSize(new THREE.Vector3());
    const maxDim = Math.max(size.x, size.y, size.z);
    gltf.scene.position.sub(center);
    camera.position.set(0, 0, maxDim * 2.2);
    modalControls.update();
    scene.add(gltf.scene);
  });

  function animate() {
    modalAnimId = requestAnimationFrame(animate);
    modalControls.update();
    modalRenderer.render(scene, camera);
  }
  animate();
}
```

```js
// ─── Mode Switching ───────────────────────────────────────────
function setViewMode(mode) {
  viewMode = mode;

  // セレクター UI 更新
  document.querySelectorAll('.vms-opt').forEach(el => {
    el.classList.toggle('active', el.dataset.mode === mode);
  });

  const canvasA = document.getElementById('canvas-plan-a');

  if (mode === 'plan-a') {
    canvasA.style.display = 'block';
    if (!planARenderer) {
      planA_init(); // 初回のみ初期化
    }
  } else {
    canvasA.style.display = 'none';
    if (planAAnimId) { cancelAnimationFrame(planAAnimId); planAAnimId = null; }
  }

  // Plan B ボタンの表示切替
  document.querySelectorAll('.btn-3d').forEach(btn => {
    btn.style.display = mode === 'plan-b' ? 'block' : 'none';
  });

  // Plan B 以外に切り替えたらモーダルを閉じる
  if (mode !== 'plan-b') closeModal3d();

  // Plan A に切り替えた場合、アニメーションを再開
  if (mode === 'plan-a' && planARenderer && !planAAnimId) {
    (function animate() {
      planAAnimId = requestAnimationFrame(animate);
      planAControls.update();
      planARenderer.render(planAScene, planACamera);
    })();
  }
}
```
